df_viz <- df %>%
  mutate(
    ethnicity = factor(ethnicity, labels = ethnicity_labels),
    ckd_diagnosed = factor(ckd_diagnosed, labels = diagnosed_labels),
    hinsu_covered = factor(hinsu_covered, labels = insurance_labels),

dashboard_theme <- theme_minimal(base_size = 14) +
  theme(
    # 1. Graph Title
    plot.title = element_text(
      family = "sans",
      face = "bold",
      size = 22,
      hjust = 0, # Left align
      margin = margin(b = 10) # Lower margin
      
    ),
    
    # 2. Graph Subtitle
    plot.subtitle = element_text(
      family = "sans",
      size = 18,
      color = "grey30",     # Grey better than black
      margin = margin(b = 10)
    ),
    
    # 3. Axis titles
    axis.title = element_text(
      family = "sans",
      face = "bold",
      size = 16
    ),
    axis.text = element_text(

      size = 14
    ),
